# Програмний код програм #
 ```c
 #include <stdio.h>

int main(int argc, char *argv[]) {
    double x;
    double y;

    scanf("%lf", &x);

    if (x <= -10.0) {
        printf("function is undefined for this value\n");
    } else if (x <= -5.0) {
        y = x * x * x - 6;
        printf("%lf\n", y);
    } else if (x <= 5.0) {
        printf("function is undefined for this value\n");
    } else if (x <= 15.0) {
        y = x * x * x - 6;
        printf("%lf\n", y);
    } else if (x < 25.0) {
        printf("function is undefined for this value\n");
    } else {
        y = 2 * x * x * x - 3 * x + 2;
        printf("%lf\n", y);
    }

    return 0;
}
```
